  title: text('title'),

  createdAt: timestamp('created_at').notNull().defaultNow(),
  updatedAt: timestamp('updated_at').notNull().defaultNow(),

  uniqueUserUrl: unique().on(table.userId, table.url),
  userIdIdx: index('links_user_id_idx').on(table.userId),

  createdIdx: index('links_created_at_idx').on(table.createdAt),

  <implementation>
    Implement as three files per the research architecture:

    domain-hints.ts:
    - DOMAIN_HINTS: Record<string, string[]> mapping known domains to guaranteed tags (github.com -> ["open-source", "code"], youtube.com -> ["video"], wikipedia.org -> ["reference", "wiki"], etc.)
    - getDomainHints(domain: string | null): string[] — matches domain against keys (supports partial match for subdomains like docs.*)

    describe.ts:
    - extractDescription(text: string): string — splits on newlines/periods, finds first paragraph with 20+ chars, truncates at sentence boundary to ~150 chars
    - Returns empty string if no meaningful text found

    classify.ts:
    - STOP_WORDS: Set<string> with ~50 common English words
    - extractKeywords(text: string, limit?: number): string[] — tokenize, filter stop words, count frequency, return top N
    - CATEGORY_SIGNALS: Record<string, string[]> with 10 categories (Technology, News, Reference, Entertainment, Shopping, Social, Education, Finance, Health, Creative) each with 10-15 signal words
    - categorize(text: string): string — score text against each category's signals, highest wins, "Other" if no score > 0
    - buildTags(text: string, domain: string | null): string[] — combine keyword extraction + domain hints, deduplicate, normalize lowercase, enforce 2-4 limit
    - enrich(opts: {bodyText, existingDescription, domain, title}): EnrichmentResult — orchestrates all three, uses existing description when present

    Import CATEGORIES from src/types/index.ts to ensure category values match the type system.
  </implementation>
