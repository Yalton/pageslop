```typescript
import { db } from '@/lib/db/client'
import { links, rawContent } from '@/lib/db/schema'
import { eq, and, desc } from 'drizzle-orm'

export async function insertLink(userId: string, url: string) {
  const domain = new URL(url).hostname.replace(/^www\./, '')
  const [link] = await db.insert(links)
    .values({ url, userId, domain, status: 'pending' })
    .returning()
  return link
}

export async function findDuplicate(userId: string, url: string) {
  const result = await db.select().from(links)
    .where(and(eq(links.userId, userId), eq(links.url, url)))
    .limit(1)
  return result[0] || null
}

export async function listLinks(userId: string) {
  return db.select().from(links)
    .where(eq(links.userId, userId))
    .orderBy(desc(links.createdAt))
}

export async function deleteLink(userId: string, linkId: string) {
  const result = await db.delete(links)
    .where(and(eq(links.id, linkId), eq(links.userId, userId)))
    .returning()
  return result[0] || null
}

export async function updateLinkMetadata(linkId: string, data: {
  title?: string | null
  description?: string | null
  ogImage?: string | null
  favicon?: string | null
  status: string
}) {
  const [updated] = await db.update(links)
    .set({ ...data, updatedAt: new Date() })
    .where(eq(links.id, linkId))
    .returning()
  return updated
}

export async function insertRawContent(linkId: string, bodyText: string | null, rawHtml: string | null) {
  await db.insert(rawContent).values({ linkId, bodyText, rawHtml })
}
```

CRITICAL: Every query that accepts a user-supplied linkId MUST include `eq(links.userId, userId)` in the WHERE clause. This prevents user A from accessing/deleting user B's links.

```typescript
import * as cheerio from 'cheerio'

export interface ScrapeResult {
  title: string
  description: string
  ogImage: string | null
  favicon: string | null
  bodyText: string | null
  rawHtml: string
}

export async function scrapeUrl(url: string): Promise<ScrapeResult> {
  const res = await fetch(url, {
    headers: { 'User-Agent': 'Mozilla/5.0 (compatible; Pageslop/1.0)' },
    signal: AbortSignal.timeout(8000), // 8s hard timeout
    redirect: 'follow',
  })

  const rawHtml = await res.text()
  const $ = cheerio.load(rawHtml)

  const title = $('meta[property="og:title"]').attr('content')
    || $('title').text().trim()
    || ''

  const description = $('meta[property="og:description"]').attr('content')
    || $('meta[name="description"]').attr('content')
    || ''

  const ogImage = $('meta[property="og:image"]').attr('content') || null

  let favicon = $('link[rel="icon"]').attr('href')
    || $('link[rel="shortcut icon"]').attr('href')
    || null

  // Resolve relative favicon URLs
  if (favicon && !favicon.startsWith('http')) {
    const base = new URL(url)
    favicon = new URL(favicon, base.origin).href
  }

  // Extract body text for Phase 2 classification
  $('script, style, nav, header, footer').remove()
  const bodyText = $('body').text().replace(/\s+/g, ' ').trim().slice(0, 5000) || null

  return { title, description, ogImage, favicon, bodyText, rawHtml }
}
```

```typescript
import PgBoss from 'pg-boss'

let boss: PgBoss | null = null

export function getBoss(): PgBoss {
  if (!boss) {
    // In API routes, use pooled connection for sending jobs only
    // The worker process creates its own PgBoss with direct connection
    boss = new PgBoss({
      connectionString: process.env.DATABASE_URL_DIRECT || process.env.DATABASE_URL!,
      noScheduling: true, // API routes only send, don't schedule
    })
    boss.on('error', console.error)
  }
  return boss
}

export async function enqueueScrapeJob(linkId: string, url: string) {
  const b = getBoss()
  // Ensure boss is started before sending
  await b.start()
  await b.send('scrape', { linkId, url }, {
    retryLimit: 2,
    retryDelay: 5,
    expireInSeconds: 60,
  })
}
```

1. Create `src/app/api/links/route.ts`:

```typescript
import { auth } from '@/lib/auth/server'
import { headers } from 'next/headers'
import { insertLink, findDuplicate, listLinks } from '@/lib/db/queries/links'
import { enqueueScrapeJob } from '@/lib/queue/boss'
import type { SaveLinksRequest, LinkWithDuplicate } from '@/types'

export async function POST(req: Request) {
  const session = await auth.api.getSession({ headers: await headers() })
  if (!session) return new Response('Unauthorized', { status: 401 })

  const body: SaveLinksRequest = await req.json()
  const urlList = body.urls
    .split('\n')
    .map((u: string) => u.trim())
    .filter((u: string) => {
      try { new URL(u); return true } catch { return false }
    })

  if (urlList.length === 0) {
    return Response.json({ error: 'No valid URLs provided' }, { status: 400 })
  }

  const saved: LinkWithDuplicate[] = []

  for (const url of urlList) {
    // Duplicate check (skip if force=true)
    if (!body.force) {
      const existing = await findDuplicate(session.user.id, url)
      if (existing) {
        saved.push({ ...existing, duplicate: true } as LinkWithDuplicate)
        continue
      }
    }

    const link = await insertLink(session.user.id, url)
    await enqueueScrapeJob(link.id, url)
    saved.push(link as LinkWithDuplicate)
  }

  return Response.json({ saved }, { status: 202 })
}

export async function GET(req: Request) {
  const session = await auth.api.getSession({ headers: await headers() })
  if (!session) return new Response('Unauthorized', { status: 401 })

  const data = await listLinks(session.user.id)
  return Response.json(data)
}
```

CRITICAL: POST returns 202 immediately. Scraping happens asynchronously via worker. Never await scrapeUrl() in the route handler.

```typescript
import { deleteLink } from '@/lib/db/queries/links'

export async function DELETE(
  req: Request,
  { params }: { params: Promise<{ id: string }> }
) {
  const session = await auth.api.getSession({ headers: await headers() })
  if (!session) return new Response('Unauthorized', { status: 401 })

  const { id } = await params
  const deleted = await deleteLink(session.user.id, id)

  if (!deleted) {
    return Response.json({ error: 'Link not found' }, { status: 404 })
  }

  return Response.json({ deleted })
}
```

3. Create `src/workers/scrape.worker.ts`:

```typescript
import PgBoss from 'pg-boss'
import { scrapeUrl } from '../lib/scraper/cheerio'
import { drizzle } from 'drizzle-orm/node-postgres'
import { Pool } from 'pg'
import { eq } from 'drizzle-orm'
import * as schema from '../lib/db/schema'

// DIRECT connection (no -pooler in hostname) — required for pg-boss LISTEN/NOTIFY
const pool = new Pool({
  connectionString: process.env.DATABASE_URL_DIRECT!,
  max: 5,
})
const db = drizzle({ client: pool, schema })

const boss = new PgBoss({
  connectionString: process.env.DATABASE_URL_DIRECT!,
})

boss.on('error', (err) => console.error('[pg-boss error]', err))

async function main() {
  await boss.start()
  console.log('[worker] pg-boss started, listening for scrape jobs...')

  await boss.work('scrape', { teamSize: 5 }, async (job) => {
    const { linkId, url } = job.data as { linkId: string; url: string }
    console.log(`[worker] scraping ${url} for link ${linkId}`)

    try {
      const result = await scrapeUrl(url)

      // Update link with scraped metadata
      await db.update(schema.links)
        .set({
          title: result.title || null,
          description: result.description || null,
          ogImage: result.ogImage,
          favicon: result.favicon,
          status: 'ready',
          updatedAt: new Date(),
        })
        .where(eq(schema.links.id, linkId))

      // Store raw content for Phase 2 classification
      await db.insert(schema.rawContent)
        .values({
          linkId,
          bodyText: result.bodyText,
          rawHtml: result.rawHtml,
        })

      console.log(`[worker] completed scraping ${url}`)
    } catch (err) {
      console.error(`[worker] failed scraping ${url}:`, err)
      await db.update(schema.links)
        .set({ status: 'error', updatedAt: new Date() })
        .where(eq(schema.links.id, linkId))
    }
  })
}

main().catch((err) => {
  console.error('[worker] fatal:', err)
  process.exit(1)
})
```

NOTE: Worker uses relative imports (not @/ alias) because it runs as a standalone Node process via tsx, not through Next.js bundler. The imports use `../lib/` paths.
