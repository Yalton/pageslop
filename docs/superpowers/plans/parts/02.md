**Interfaces:**
- Consumes:
  - `authenticateBearer(req: Request): Promise<{ userId: string } | null>` from `src/lib/auth/api-token.ts` (Task 2)
  - `stashTabs(userId: string, tabs: { url: string; title?: string; closedAt: Date }[]): Promise<{ saved: string[]; inserted: { id: string; url: string }[] }>` from `src/lib/db/queries/links.ts` (Task 3). It deduplicates repeated URLs within one batch and returns the client's URL strings unchanged in `saved`.
  - `enqueueScrapeJob(linkId: string, url: string, importJobId?: string): Promise<void>` from `src/lib/queue/boss.ts` (existing)
- Produces:
  - `POST /api/stash`: body `{ tabs: { url: string; title?: string | null; favicon?: string | null; closedAt?: unknown }[] }` (max 100). Responses: `200 { saved: string[]; rejected: { url: string; reason: 'invalid_url' | 'unsupported_protocol' }[] }`, `400 { error }` for a malformed body or more than 100 tabs, `401 { error: 'Unauthorized' }`.
  - `GET /api/stash/ping`: `200 { ok: true, user: { email: string } }` or `401 { error: 'Unauthorized' }`.
  - `getUserEmail(userId: string): Promise<string | null>` in `src/lib/db/queries/users.ts`.
  - `/api/stash` and everything under it bypass the session-cookie redirect in `src/middleware.ts`.

Behaviour notes for the implementer:
- A URL string that does not parse, or that parses with a protocol other than `http:`/`https:`, is listed in `rejected` and is not saved. It does not fail the request. A `url` that is not a string at all is a malformed body and returns 400.
- The route never trims, normalises or re-serialises URL strings. `new URL()` is only used to check the protocol. The exact string the client sent goes to `stashTabs` and comes back in `saved` (Review Focus #1).
- A missing, non-string or unparseable `closedAt` falls back to the server's current time. A `closedAt` later than the server's now is clamped to now, so client clock skew cannot postpone expiry (Review Focus #5).
- Scrape enqueue failures are logged and swallowed. The links are already saved, so the response still lists them in `saved`.

```ts
/**
 * POST /api/stash — save closing browser tabs (bearer token auth).
 *
 * The extension closes a tab only if its URL string comes back in `saved`,
 * so URL strings are never trimmed or normalised here. `new URL()` is used
 * only to check the protocol.
 */

import { z } from 'zod'
import { authenticateBearer } from '@/lib/auth/api-token'
import { stashTabs } from '@/lib/db/queries/links'
import { enqueueScrapeJob } from '@/lib/queue/boss'

const MAX_TABS = 100

const stashBodySchema = z.object({
  tabs: z
    .array(
      z.object({
        url: z.string(),
        title: z.string().nullish(),
        favicon: z.string().nullish(),
        closedAt: z.unknown().optional(),
      }),
    )
    .max(MAX_TABS),
})

type RejectReason = 'invalid_url' | 'unsupported_protocol'

function checkUrl(raw: string): RejectReason | null {
  let parsed: URL
  try {
    parsed = new URL(raw)
  } catch {
    return 'invalid_url'
  }
  if (parsed.protocol !== 'http:' && parsed.protocol !== 'https:') {
    return 'unsupported_protocol'
  }
  return null
}

/** Invalid, missing or future closedAt values fall back to the server's now. */
function parseClosedAt(value: unknown, now: Date): Date {
  if (typeof value !== 'string') return now
  const parsed = new Date(value)
  if (Number.isNaN(parsed.getTime())) return now
  if (parsed.getTime() > now.getTime()) return now
  return parsed
}

export async function POST(req: Request): Promise<Response> {
  const authed = await authenticateBearer(req)
  if (!authed) return Response.json({ error: 'Unauthorized' }, { status: 401 })

  let raw: unknown
  try {
    raw = await req.json()
  } catch {
    return Response.json({ error: 'Invalid JSON body' }, { status: 400 })
  }

  const parsed = stashBodySchema.safeParse(raw)
  if (!parsed.success) {
    return Response.json(
      { error: `Invalid body: expected { tabs: [{ url, title?, closedAt? }] } with at most ${MAX_TABS} tabs` },
      { status: 400 },
    )
  }

  const now = new Date()
  const rejected: { url: string; reason: RejectReason }[] = []
  const valid: { url: string; title?: string; closedAt: Date }[] = []

  for (const tab of parsed.data.tabs) {
    const reason = checkUrl(tab.url)
    if (reason) {
      rejected.push({ url: tab.url, reason })
      continue
    }
    valid.push({
      url: tab.url,
      title: tab.title ?? undefined,
      closedAt: parseClosedAt(tab.closedAt, now),
    })
  }

  if (valid.length === 0) {
    return Response.json({ saved: [], rejected })
  }

  let result: Awaited<ReturnType<typeof stashTabs>>
  try {
    result = await stashTabs(authed.userId, valid)
  } catch (err) {
    console.error('[stash] stashTabs failed', err)
    return Response.json({ error: 'Failed to save tabs' }, { status: 500 })
  }

  // Links are already saved. A queue failure must not hide that from the
  // extension, or it would keep the tabs open and re-send them forever.
  for (const link of result.inserted) {
    try {
      await enqueueScrapeJob(link.id, link.url)
    } catch (err) {
      console.error('[stash] enqueueScrapeJob failed', link.id, err)
    }
  }

  return Response.json({ saved: result.saved, rejected })
}
```

```ts
/**
 * Browser extension API tokens.
 *
 * GET  /api/settings/tokens — list the caller's tokens (never hashes)
 * POST /api/settings/tokens — create a token; the raw value is returned once
 *
 * Session auth. Revoking lives in ./[id]/route.ts.
 */

import { z } from 'zod'
import { auth } from '@/lib/auth/server'
import { headers } from 'next/headers'
import { generateToken } from '@/lib/auth/api-token'
import { createApiToken, listApiTokens } from '@/lib/db/queries/api-tokens'

const createTokenSchema = z.object({
  name: z.string().trim().min(1).max(100),
})

export async function GET(): Promise<Response> {
  const session = await auth.api.getSession({ headers: await headers() })
  if (!session) return new Response('Unauthorized', { status: 401 })

  const rows = await listApiTokens(session.user.id)
  // Pick fields explicitly so a hash can never leak, whatever the query selects
  const tokens = rows.map((row) => ({
    id: row.id,
    name: row.name,
    createdAt: row.createdAt,
    lastUsedAt: row.lastUsedAt ?? null,
  }))
  return Response.json({ tokens })
}

export async function POST(req: Request): Promise<Response> {
  const session = await auth.api.getSession({ headers: await headers() })
  if (!session) return new Response('Unauthorized', { status: 401 })

  let raw: unknown
  try {
    raw = await req.json()
  } catch {
    return Response.json({ error: 'Invalid request body' }, { status: 400 })
  }

  const parsed = createTokenSchema.safeParse(raw)
  if (!parsed.success) {
    return Response.json({ error: 'name is required (1 to 100 characters)' }, { status: 400 })
  }

  const { raw: token, hash } = generateToken()
  const created = await createApiToken(session.user.id, parsed.data.name, hash)

  return Response.json({ id: created.id, name: created.name, token }, { status: 201 })
}
```

```ts
/**
 * DELETE /api/settings/tokens/[id] — revoke one of the caller's tokens.
 * Session auth. Deleting the row is the revocation; bearer requests using
 * that token get 401 from then on.
 */

import { auth } from '@/lib/auth/server'
import { headers } from 'next/headers'
import { deleteApiToken } from '@/lib/db/queries/api-tokens'

export async function DELETE(
  req: Request,
  { params }: { params: Promise<{ id: string }> },
): Promise<Response> {
  const session = await auth.api.getSession({ headers: await headers() })
  if (!session) return new Response('Unauthorized', { status: 401 })

  const { id } = await params
  const deleted = await deleteApiToken(session.user.id, id)
  if (!deleted) {
    return Response.json({ error: 'Token not found' }, { status: 404 })
  }

  return Response.json({ deleted: true })
}
```

- [ ] **Step 4: Write the bulk route**

Create `src/app/api/links/bulk-stash-state/route.ts`:

```ts
/**
 * POST /api/links/bulk-stash-state — set stashState on many links at once
 * (Open all / Keep all / Dismiss all on /dashboard/revisit).
 *
 * Session auth. bulkSetStashState filters by the caller's user id, so ids
 * belonging to other users are silently skipped and not counted.
 */

import { z } from 'zod'
import { auth } from '@/lib/auth/server'
import { headers } from 'next/headers'
import { bulkSetStashState } from '@/lib/db/queries/links'
import { STASH_STATES } from '@/types'

const MAX_IDS = 500

const bulkSchema = z.object({
  ids: z.array(z.string().min(1)).min(1).max(MAX_IDS),
  stashState: z.enum(STASH_STATES),
})

export async function POST(req: Request): Promise<Response> {
  const session = await auth.api.getSession({ headers: await headers() })
  if (!session) return new Response('Unauthorized', { status: 401 })

  let raw: unknown
  try {
    raw = await req.json()
  } catch {
    return Response.json({ error: 'Invalid request body' }, { status: 400 })
  }

  const parsed = bulkSchema.safeParse(raw)
  if (!parsed.success) {
    return Response.json(
      { error: `Expected { ids: string[] (1-${MAX_IDS}), stashState: ${STASH_STATES.join(' | ')} }` },
      { status: 400 },
    )
  }

  const updated = await bulkSetStashState(session.user.id, parsed.data.ids, parsed.data.stashState)
  return Response.json({ updated })
}
```

Next.js matches the static `bulk-stash-state` segment before the dynamic `[id]` segment, so this route does not collide with `/api/links/[id]`.
