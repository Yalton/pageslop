```ts
import type { LinkWithTags } from '@/types'

/**
 * Client-side mirror of STASH_EXPIRE_DAYS. The server value lives in
 * src/lib/stash/expiry.ts, which reads process.env and must not be bundled
 * into client components. Keep this in sync with the server default.
 */
export const CLIENT_STASH_EXPIRE_DAYS = 14
/** Rows show "expires in Nd" once this many days or fewer remain. */
export const EXPIRY_WARNING_DAYS = 3
/** "Dismiss all" asks for confirmation above this many items. */
export const DISMISS_CONFIRM_THRESHOLD = 10

const DAY_MS = 86_400_000

export function isRevisitable(link: Pick<LinkWithTags, 'stashState'>): boolean {
  return link.stashState === 'inbox' || link.stashState === 'read'
}

export function selectRevisitLinks(links: LinkWithTags[]): LinkWithTags[] {
  return links.filter(isRevisitable)
}

export function countInbox(links: Pick<LinkWithTags, 'stashState'>[]): number {
  let count = 0
  for (const link of links) {
    if (link.stashState === 'inbox') count++
  }
  return count
}

export function daysSince(date: Date | string, now: Date): number {
  const elapsed = now.getTime() - new Date(date).getTime()
  return Math.max(0, Math.floor(elapsed / DAY_MS))
}

export function expiresInDays(
  stashedAt: Date | string,
  now: Date,
  days: number = CLIENT_STASH_EXPIRE_DAYS,
): number {
  const expiresAt = new Date(stashedAt).getTime() + days * DAY_MS
  return Math.max(0, Math.ceil((expiresAt - now.getTime()) / DAY_MS))
}

export interface RevisitRowMeta {
  stashedLabel: string | null
  expiryLabel: string | null
}

export function rowMeta(
  stashedAt: Date | string | null,
  now: Date,
  days: number = CLIENT_STASH_EXPIRE_DAYS,
): RevisitRowMeta {
  if (!stashedAt) return { stashedLabel: null, expiryLabel: null }
  const remaining = expiresInDays(stashedAt, now, days)
  return {
    stashedLabel: `stashed ${daysSince(stashedAt, now)}d ago`,
    expiryLabel: remaining <= EXPIRY_WARNING_DAYS ? `expires in ${remaining}d` : null,
  }
}

export function needsDismissConfirm(count: number): boolean {
  return count > DISMISS_CONFIRM_THRESHOLD
}

export interface OpenAllResult {
  openedIds: string[]
  blockedCount: number
}

/**
 * Opens each URL with `openFn`. A null/undefined return means the browser
 * blocked the pop-up. Only ids whose tab actually opened are returned, so the
 * caller marks just those as read.
 */
export function openAllUrls(
  items: { id: string; url: string }[],
  openFn: (url: string) => unknown,
): OpenAllResult {
  const openedIds: string[] = []
  let blockedCount = 0
  for (const item of items) {
    const handle = openFn(item.url)
    if (handle == null) {
      blockedCount++
    } else {
      openedIds.push(item.id)
    }
  }
  return { openedIds, blockedCount }
}
```

```ts
export function isNavActive(pathname: string, href: string): boolean {
  if (href === '#') return false
  if (pathname === href) return true
```

```tsx
type GroupMode = 'domain' | 'topic'

/**
 * Opens a URL in a new tab and returns the window handle, or null when the
 * browser blocked the pop-up. `noopener` is deliberately NOT passed as a
 * window feature: with it, window.open always returns null and blocked tabs
 * become indistinguishable from opened ones. The opener is cut manually instead.
 */
function openTab(url: string): Window | null {
  const handle = window.open(url, '_blank')
  if (handle) handle.opener = null
  return handle
}

export function RevisitView() {
  const { links, isLoading, setStashState } = useLinks()
  const [mode, setMode] = useState<GroupMode>('domain')
  const [query, setQuery] = useState('')

  const revisitLinks = useMemo(() => selectRevisitLinks(links), [links])
  const visible = useMemo(() => filterLinks(revisitLinks, query, null), [revisitLinks, query])
  const groups = useMemo(
    () => (mode === 'domain' ? groupByDomain(visible) : groupLinks(visible)),
    [mode, visible],
  )

  async function applyState(ids: string[], state: StashState, successMessage?: string) {
    try {
      await setStashState(ids, state)
      if (successMessage) toast.success(successMessage)
    } catch {
      toast.error('Could not update links. Please try again.')
    }
  }

  function handleOpenAll(groupLinksList: LinkWithTags[]) {
    const { openedIds, blockedCount } = openAllUrls(
      groupLinksList.map((l) => ({ id: l.id, url: l.url })),
      openTab,
    )
    const toMark = openedIds.filter((id) => groupLinksList.find((l) => l.id === id)?.stashState === 'inbox')

    if (blockedCount > 0) {

      )

    if (
      needsDismissConfirm(groupLinksList.length) &&
      !window.confirm(`Dismiss all ${groupLinksList.length} links in ${name}? They will be archived.`)
    ) {
      return
    }
    void applyState(
      groupLinksList.map((l) => l.id),
      'archived',
      `Dismissed ${groupLinksList.length} link${groupLinksList.length === 1 ? '' : 's'}`,
    )

            Browser extension settings
          </Link>
          .
        </p>
      </div>
    )
  }

          <SearchInput value={query} onChange={setQuery} />

            size="sm"
            pressed={mode === 'domain'}
            onPressedChange={() => setMode('domain')}
            aria-label="Group by domain"
          >
            Domain
          </Toggle>
          <Toggle
            size="sm"
            pressed={mode === 'topic'}
            onPressedChange={() => setMode('topic')}
            aria-label="Group by topic"
```

```ts
import { format } from 'date-fns'

/** Row shape returned by GET /api/settings/tokens. Dates are ISO strings over JSON. */
export interface ApiTokenSummary {
  id: string
  name: string
  createdAt: string
  lastUsedAt: string | null
}

/** Response of POST /api/settings/tokens. `token` is the raw value, shown once. */
export interface CreatedApiToken {
  id: string
  name: string
  token: string
}

export const TOKEN_NAME_MAX = 100

export function formatTokenDate(value: string | Date | null): string {
  if (value === null) return 'Never'
  const date = new Date(value)
  if (Number.isNaN(date.getTime())) return 'Never'
  return format(date, 'MMM d, yyyy')
}

/** Returns an error message, or null when the name is acceptable. */
export function validateTokenName(name: string): string | null {
  const trimmed = name.trim()
  if (trimmed.length === 0) return 'Name is required'
  if (trimmed.length > TOKEN_NAME_MAX) return `Name must be ${TOKEN_NAME_MAX} characters or fewer`
  return null
}
```

```tsx
'use client'

import { useState } from 'react'
import useSWR from 'swr'
import { AlertCircle, Check, Copy, Puzzle, Trash2 } from 'lucide-react'
import { toast } from 'sonner'
import { Button } from '@/components/ui/button'
import {
  formatTokenDate,
  validateTokenName,
  TOKEN_NAME_MAX,
  type ApiTokenSummary,
  type CreatedApiToken,
} from '@/lib/settings/token-format'

const TOKENS_URL = '/api/settings/tokens'
const README_EXTENSION_URL = 'https://github.com/Citreum-LLC/pageslop#firefox-extension'

async function fetchTokens(url: string): Promise<ApiTokenSummary[]> {
  const res = await fetch(url)
  if (!res.ok) throw new Error('Failed to load tokens')
  return res.json()
}

const inputClass =
  'flex-1 min-w-[240px] rounded-md border border-input bg-background px-3 py-2 text-sm ring-offset-background placeholder:text-muted-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring focus-visible:ring-offset-2'

export function ExtensionTokensCard() {
  const { data: tokens, error: loadError, isLoading, mutate } = useSWR<ApiTokenSummary[]>(TOKENS_URL, fetchTokens)
  const [name, setName] = useState('')
  const [creating, setCreating] = useState(false)
  const [formError, setFormError] = useState<string | null>(null)
  const [created, setCreated] = useState<CreatedApiToken | null>(null)
  const [copied, setCopied] = useState(false)
  const [revokingId, setRevokingId] = useState<string | null>(null)

  async function handleCreate(e: React.FormEvent) {
    e.preventDefault()
    const validation = validateTokenName(name)
    if (validation) {
      setFormError(validation)
      return
    }
    setFormError(null)
    setCreating(true)
    try {
      const res = await fetch(TOKENS_URL, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ name: name.trim() }),
      })
      if (!res.ok) {
        const body = (await res.json().catch(() => null)) as { error?: string } | null
        setFormError(body?.error ?? 'Could not create token')
        return
      }
      const token = (await res.json()) as CreatedApiToken
      setCreated(token)
      setCopied(false)
      setName('')
      await mutate()
    } catch {
      setFormError('Could not create token')
    } finally {
      setCreating(false)
    }
  }

  async function handleCopy() {
    if (!created) return
    try {
      await navigator.clipboard.writeText(created.token)
      setCopied(true)
      toast.success('Token copied')
    } catch {
      toast.error('Copy failed. Select the token and copy it manually.')
    }
  }

  async function handleRevoke(token: ApiTokenSummary) {
    if (!window.confirm(`Revoke "${token.name}"? Extensions using it will stop syncing.`)) return
    setRevokingId(token.id)
    try {
      const res = await fetch(`${TOKENS_URL}/${token.id}`, { method: 'DELETE' })
      if (!res.ok && res.status !== 404) throw new Error('revoke failed')
      if (created?.id === token.id) setCreated(null)
      toast.success(`Revoked "${token.name}"`)
      await mutate()
    } catch {
      toast.error('Could not revoke token')
    } finally {
      setRevokingId(null)
    }
  }

  return (
    <>
      <h2 id="browser-extension" className="text-lg font-semibold mt-8 mb-4 scroll-mt-6">
        <span className="inline-flex items-center gap-2">
          <Puzzle className="size-5" />
          Browser Extension
        </span>
      </h2>

      <div className="bg-card rounded-lg border p-6">
        <h3 className="font-semibold mb-1">Extension tokens</h3>
        <p className="text-sm text-muted-foreground mb-4">
          The Firefox extension uses a token to save idle tabs to your To Revisit list. Create one token per browser.
        </p>

        <form onSubmit={handleCreate} className="flex flex-wrap gap-2">
          <input
            type="text"
            value={name}
            onChange={(e) => setName(e.target.value)}
            placeholder="Token name, e.g. Firefox laptop"
            maxLength={TOKEN_NAME_MAX}
            aria-label="Token name"
            className={inputClass}
          />
          <Button type="submit" disabled={creating}>
            {creating ? 'Creating...' : 'Create token'}
          </Button>
        </form>

        {formError && (
          <div className="mt-3 flex items-center gap-2 text-sm text-destructive">
            <AlertCircle className="size-4" />
            {formError}
          </div>
        )}

        {created && (
          <div className="mt-4 rounded-md border border-primary/40 bg-primary/5 p-4">
            <p className="text-sm font-medium mb-2">
              Token &ldquo;{created.name}&rdquo; created. Copy it now, you won&apos;t see this again.
            </p>
            <div className="flex flex-wrap gap-2">
              <input
                type="text"
                readOnly
                value={created.token}
                onFocus={(e) => e.currentTarget.select()}
                aria-label="New token"
                className={`${inputClass} font-mono`}
              />
              <Button type="button" variant="outline" onClick={handleCopy}>
                {copied ? <Check className="size-4" /> : <Copy className="size-4" />}
                {copied ? 'Copied' : 'Copy'}
              </Button>
              <Button type="button" variant="ghost" onClick={() => setCreated(null)}>
                Done
              </Button>
            </div>
          </div>
        )}

        <div className="mt-6">
          <h4 className="text-sm font-semibold mb-2">Active tokens</h4>
          {isLoading && <p className="text-sm text-muted-foreground">Loading...</p>}
          {loadError && (
            <div className="flex items-center gap-2 text-sm text-destructive">
              <AlertCircle className="size-4" />
              Could not load tokens
            </div>
          )}
          {tokens && tokens.length === 0 && (
            <p className="text-sm text-muted-foreground">No tokens yet.</p>
          )}
          {tokens && tokens.length > 0 && (
            <ul className="flex flex-col divide-y border rounded-md">
              {tokens.map((token) => (
                <li key={token.id} className="flex flex-wrap items-center gap-x-4 gap-y-1 px-3 py-2">
                  <span className="text-sm font-medium flex-1 min-w-[120px] truncate">{token.name}</span>
                  <span className="text-xs text-muted-foreground">
                    Created {formatTokenDate(token.createdAt)}
                  </span>
                  <span className="text-xs text-muted-foreground">
                    Last used {formatTokenDate(token.lastUsedAt)}
                  </span>
                  <Button
                    type="button"
                    variant="ghost"
                    size="sm"
                    className="text-destructive hover:text-destructive"
                    disabled={revokingId === token.id}
                    onClick={() => handleRevoke(token)}
                  >
                    <Trash2 className="size-4" />
                    {revokingId === token.id ? 'Revoking...' : 'Revoke'}
                  </Button>
                </li>
              ))}
            </ul>
          )}
        </div>

        <div className="mt-6">
          <h4 className="text-sm font-semibold mb-2">Setup</h4>
          <ol className="list-decimal pl-5 text-sm text-muted-foreground space-y-1">
            <li>
              Install the extension by following the{' '}
              <a
                href={README_EXTENSION_URL}
                target="_blank"
                rel="noopener noreferrer"
                className="underline underline-offset-4 hover:text-foreground"
              >
                Firefox extension section of the README
              </a>
              .
            </li>
            <li>Create a token above and copy it.</li>
            <li>
              Open the extension&apos;s settings. Enter this server&apos;s URL (the address in your browser bar, without{' '}
              <code className="font-mono">/dashboard</code>) and paste the token.
            </li>
            <li>Click Save. The extension checks the connection and shows whether it worked.</li>
          </ol>
        </div>
      </div>
    </>
  )
}
```
